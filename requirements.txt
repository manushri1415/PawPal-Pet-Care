# Runtime dependencies only -- this is exactly what the production image
# installs (see Dockerfile). Test tooling lives in requirements-dev.txt; for
# local development and CI, install that file instead (it includes this one).
#
# Pinned to exact versions. Unpinned `>=` ranges let every fresh install --
# CI, a Docker build, a Lambda package -- resolve newer majors than the ones
# the suite last passed against (anthropic 0.x -> 1.x moved its HTTP client
# from httpx to httpx2; starlette 1.3 -> 1.6 between two installs a day apart).
# These are the versions the full test suite and the evaluation harness pass
# on, resolved in a clean virtualenv on Python 3.13. Bump deliberately: change
# a pin, reinstall into a fresh venv, rerun both.

boto3==1.43.94            # imported only by the DynamoDB backend; SQLite needs none of it

# NOTE ON RETRIEVAL:
# We deliberately use a dependency-light, pure-Python vector store (see
# pawpal_ai/vectorstore.py) with local feature-hashing embeddings. This keeps the
# project reproducible on any machine WITHOUT a C++ compiler. A heavier vector DB
# (e.g. Chroma) needs to build native wheels (chroma-hnswlib -> MSVC), which fails
# on stock Windows/Python 3.13 and hurts "runs on a grader's machine". The
# VectorStore interface is swappable if you later want a neural embedding backend.
